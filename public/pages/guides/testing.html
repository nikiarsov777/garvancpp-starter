#include "crow.h"
#include "doctest.h"

TEST_CASE("hello returns greeting") {
    crow::SimpleApp app;
    CROW_ROUTE(app, "/")([] { return "hi"; });

    crow::request req;
    req.url = "/";
    crow::response res;
    app.handle(req, res);

    CHECK(res.code == 200);
    CHECK(res.body == "hi");
}
