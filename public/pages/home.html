#include "crow.h"

int main()
{
    crow::SimpleApp app;

    CROW_ROUTE(app, "/")([]{
        return "Hello from Garvan!";
    });

    app.port(9090).multithreaded().run();
}
CROW_ROUTE(app, "/api/users/<int>")
([](int id) {
    crow::json::wvalue user;
    user["id"]    = id;
    user["name"]  = "Ada";
    user["email"] = "ada@example.com";
    return user;
});
