Modelos de projeto

Em vez de construir tudo do zero, pode começar de um skeleton de projeto Garvan. A estrutura recomendada corresponde ao que as ferramentas kalpasan CLI e garvan-migrate incluídas esperam encontrar.

Estrutura de diretórios

my-app/
├── app/
│   ├── controllers/        # request handlers, uma classe por recurso
│   ├── models/             # modelos de BD que estendem OModel
│   └── services/           # lógica de negócio entre controllers e models
├── db/
│   ├── migrations/         # ficheiros SQL corridos pelo garvan-migrate
│   └── seeders/            # dados iniciais
├── public/                 # templates mustache (HTML)
├── static/                 # servido em /static/* (css, js, images)
├── routes/
│   ├── ApiRoutes.cpp       # endpoints JSON / REST
│   └── WebRoutes.cpp       # páginas HTML
├── vendors/Garvan/         # libgarvan.a + headers incluídos
├── .env                    # configuração de ambiente
├── CMakeLists.txt
└── main.cpp                # entry point: liga as rotas

Modelo CMake

Um CMakeLists.txt mínimo seguindo a estrutura acima:

cmake_minimum_required(VERSION 3.20)
project(MyApp CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${PROJECT_SOURCE_DIR}/bin)

file(GLOB_RECURSE SOURCES
    "main.cpp"
    "routes/*.cpp"
    "app/controllers/*.cpp"
    "app/models/*.cpp"
    "app/services/*.cpp"
)

add_executable(app_bin ${SOURCES})

target_include_directories(app_bin PRIVATE
    .
    vendors/Garvan
    vendors/Garvan/include
)

target_link_libraries(app_bin PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/vendors/Garvan/libgarvan.a
    pthread
)

set_target_properties(app_bin PROPERTIES OUTPUT_NAME app.bin)

A partir daí, o seu main.cpp só precisa ligar as rotas:

#include "vendors/Garvan/crow.h"
#include "routes/ApiRoutes.h"
#include "routes/WebRoutes.h"

using namespace Routes;

int main()
{
    crow::SimpleApp app;
    crow::mustache::set_global_base("public/");

    WebRoutes  webRoutes(app);
    ApiRoutes  apiRoutes(app);

    app.port(9090).multithreaded().run();
}
